D:
SELECT DEPARTMENT,
SUM(SALARY) AS TOTAL_SALARY,
AVG(SALARY) AS AVERAGE_SALARY,
MAX(SALARY) AS MAXIMUM_SALARY,
MIN(SALARY) AS MINIMUM_SALARY
FROM EMPLOYEE
GROUP BY DEPARTMENT									    
HAVING SUM(SALARY)>30000;									      

OUTPUT:

E:
SELECT DEPARTMENT,COUNT(*)
AS NO_OF_EMPLOYEES
FROM EMPLOYEE
GROUP BY DEPARTMENT
HAVING COUNT(*)>2;

OUTPUT:
DEPARTMENT      NO_OF_EMPLOYEES
--------------- ---------------
IT                            4
ACCOUNTS                      3
SELLS                         3


CREATE TABLE EMPLOYEE
(EMPNO NUMBER(3) PRIMARY KEY,NAME VARCHAR2(15),DESIGNATION VARCHAR2(15) NOT NULL,DEPARTMENT VARCHAR2(15) NOT NULL,
GENDER CHAR(1) NOT NULL,SALARY NUMBER(10,2) NOT NULL,
CHECK(GENDER IN('M','F')),
CHECK(DEPARTMENT IN('SELLS','ACCOUNTS','IT')),
CHECK((
DEPARTMENT='SELLS' AND DESIGNATION IN('CLERK','MANAGER','SUPERVISOR'))
OR(
DEPARTMENT='ACCOUNTS' AND DESIGNATION IN('ACCOUNTANT','MANAGER','SUPERVISOR'))
OR(
DEPARTMENT='IT' AND DESIGNATION IN('ANALYST','MANAGER','SUPERVISOR'))
));

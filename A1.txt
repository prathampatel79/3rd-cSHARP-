using System;
using System.Collections.Generic;
using System.Text;
namespace A1

class Program{
static void Main(string[ args)

Console.Write("Enter the number of strings:"); int num = Convert.Tolnt32(Console.ReadLine());

string[] str = new string[num];

int[] vcount = new int[num];

int max = 0;

for (int i = 0; i < num; i++)

Console.Write("Enter the strings:" + (i + 1) + ":"); str[i] = Console.ReadLine();

int count = 0;

foreach (char ch in str[i])

char c = char.ToUpper(ch);

if (c == 'A' || c == 'E' || c == 'l' || c == 'O' || c == 'U')

count++;

}

vcount[i] = count;

if (count > max)

max = count;

Console.WriteLine("\nString with highest vowels are:") for (int j = 0; j < num; j++)

{

if (vcount[j] == max)

Console.WriteLine(str[j]);

}

Console.Write("Number of vowels:" + max); Console.ReadKey();

}
